# --- Voice stack: PIN THESE TOGETHER ---------------------------------------
# discord.py 2.7.1 talks to davey via the 0.1.x API: gateway.py calls
# process_commit()/process_welcome(), which install the MLS group keys.
# davey 0.2.0 (released 2026-11-02) moved key installation behind
# prepare_transition()/execute_transition() — methods discord.py 2.7.1 never
# calls. With 0.2.0 the DAVE handshake completes but no keys are ever set, so
# the server cannot decrypt the bot's frames: the bot joins voice, shows the
# speaking glow, sends well-formed RTP… and nobody hears it.
# Do NOT upgrade these two independently — bump both together only once
# discord.py ships support for the 0.2.x transition API.
discord.py[voice]==2.7.1
davey==0.1.7
# [voice] already requires PyNaCl>=1.5,<1.6; pinned to the validated one.
PyNaCl==1.5.0
